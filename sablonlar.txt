package com.genixo.studyscore.domain.entity.exam.question;

import lombok.AllArgsConstructor;
import lombok.Data;
import lombok.NoArgsConstructor;

import java.io.Serial;
import java.io.Serializable;

@Data
@AllArgsConstructor
@NoArgsConstructor
public class ScoringConfig implements Serializable {
    @Serial
    private static final long serialVersionUID = 1L;

    private ScoringStrategy strategy = ScoringStrategy.BINARY;
    private Boolean allowPartialCredit = false;
    private Double penaltyPerWrong = 0.0; // 0.0 - 1.0 (yanlış cevap başına puan kesintisi oranı)
    private Boolean roundScore = false; // True ise 0.5 → 1.0 or 0.0
    private Integer decimalPlaces = 2; // Score precision
}

package com.genixo.studyscore.domain.entity.exam.question;


public enum ScoringStrategy {
    BINARY,              // All or nothing (True/False, Multiple Choice)
    PROPORTIONAL,        // Partial credit (Multiple Response, Fill Blanks, Matching)
    POSITION_BASED,      // Position-based scoring (Ordering)
    MANUAL,              // Human grading (Essay, Audio, Video, Image)
    HYBRID               // AI + Human (Essay with AI suggestions)
}
